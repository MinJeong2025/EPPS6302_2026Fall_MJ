Part 5c

Collect comments with collect_comments() from two to four videos per channel, at most 300 comments per video. Write the collection log and the manifest (Part 5d). Record for each video the comment count it reports (get_video_stats()) and the number you received

i. Before you join comments to video titles, make sure each video appears once and each comment

once: distinct(comment_id, .keep_all = TRUE) and left_join(..., relationship = "many-to-one"). A repeated video otherwise copies its comments, and corpus() stops with docnames must be unique.

library(tidyverse)
library(tuber)
library(jsonlite)

dir_yt <- "data/youtube"
dir.create(dir_yt, recursive = TRUE, showWarnings = FALSE)

cache_file <- file.path(dir_yt, "pre_comments_bundle.rds")

if (!file.exists(cache_file)) {

  candidates <- channel_uploads |>
    filter(kept) |>
    select(video_id, channel, title, date) |>
    distinct(video_id, .keep_all = TRUE)

  stats <- get_video_stats(
    candidates$video_id,
    include_content_details = TRUE
  ) |>
    tibble::as_tibble() |>
    distinct(video_id, .keep_all = TRUE)

  video_info <- candidates |>
    left_join(
      stats |> select(video_id, view_count, comment_count),
      by = "video_id",
      relationship = "many-to-one"
    )

  targets <- video_info |>
    filter(!is.na(comment_count), comment_count > 0) |>
    group_by(channel) |>
    slice_max(view_count, n = 3, with_ties = FALSE) |>
    ungroup() |>
    distinct(video_id, .keep_all = TRUE)

  print(targets |> select(channel, title, comment_count))

  stopifnot(
    nrow(targets) == 9,
    !anyDuplicated(targets$video_id)
  )

  collect_comments <- function(video_id, n_max = 300) {

    status <- "ok"

    out <- tryCatch(
      withCallingHandlers(
        get_all_comments(video_id, max_results = n_max),

        tuber_comments_disabled = function(w) {
          status <<- "disabled"
          invokeRestart("muffleWarning")
        },

        tuber_no_comments = function(w) {
          if (status == "ok") status <<- "none"
          invokeRestart("muffleWarning")
        }
      ),
      error = function(e) e
    )

    Sys.sleep(1)

    if (inherits(out, "error")) {
      return(list(
        data = NULL,
        log = tibble(
          video_id,
          status = "error",
          n = 0L,
          message = conditionMessage(out)
        )
      ))
    }

    list(
      data = tibble::as_tibble(out),
      log = tibble(
        video_id,
        status,
        n = nrow(out),
        message = NA_character_
      )
    )
  }

  runs <- purrr::map(
    targets$video_id,
    collect_comments,
    n_max = 300,
    .progress = TRUE
  )

  yt_log <- purrr::map(runs, "log") |>
    purrr::list_rbind() |>
    mutate(
      collected_at = format(Sys.time(), tz = "UTC", usetz = TRUE)
    )

  yt_comments <- purrr::map(runs, "data") |>
    purrr::list_rbind()

  if (nrow(yt_comments) > 0) {

    yt_comments <- yt_comments |>
      distinct(comment_id, .keep_all = TRUE) |>
      left_join(
        targets |>
          distinct(video_id, .keep_all = TRUE) |>
          transmute(
            video_id,
            channel_title = channel,
            video_title = title
          ),
        by = "video_id",
        relationship = "many-to-one"
      )
  }

  received_counts <- if (nrow(yt_comments) > 0) {
    yt_comments |>
      count(video_id, name = "received")
  } else {
    tibble(video_id = character(), received = integer())
  }

  per_video <- targets |>
    select(video_id, channel, title, comment_count) |>
    rename(reported = comment_count) |>
    left_join(
      received_counts,
      by = "video_id",
      relationship = "one-to-one"
    ) |>
    mutate(received = coalesce(received, 0L))

  readr::write_csv(
    yt_log,
    file.path(dir_yt, "pre_collection_log.csv")
  )

  readr::write_csv(
    per_video,
    file.path(dir_yt, "pre_video_counts.csv")
  )

  saveRDS(
    yt_comments,
    file.path(dir_yt, "pre_comments.rds")
  )

  manifest <- list(
    source = "YouTube Data API v3 via tuber",
    tuber = as.character(packageVersion("tuber")),
    credential = "API key (public data only)",
    collected_at = format(Sys.time(), tz = "UTC", usetz = TRUE),

    search = list(
      term = "US midterm election",
      orders = c("relevance", "viewCount"),
      published_after = "2026-09-01T00:00:00Z",
      published_before = "2026-10-28T00:00:00Z"
    ),

    channels = c(
      "NBC News",
      "CNN",
      "FOX 9 Minneapolis-St. Paul"
    ),

    title_filter = title_pattern,
    selection = "Three highest-viewed eligible videos per channel",
    max_comments_per_video = 300,
    n_videos = nrow(targets),
    n_comments = nrow(yt_comments),

    retention = "Delete or refresh within 30 days",
    delete_by = as.character(Sys.Date() + 30)
  )

  jsonlite::write_json(
    manifest,
    file.path(dir_yt, "pre_manifest.json"),
    auto_unbox = TRUE,
    pretty = TRUE
  )


  saveRDS(
    list(
      targets = targets,
      comments = yt_comments,
      log = yt_log,
      per_video = per_video
    ),
    cache_file
  )
}

summarise_comments <- function(df) {
  n <- nrow(df)
  likes <- df$likes

  top_1_n <- max(1, ceiling(n * 0.01))
  top_1_likes_share <- if (sum(likes) > 0) {
    100 * sum(head(sort(likes, decreasing = TRUE), top_1_n)) / sum(likes)
  } else {
    NA_real_
  }

  authors <- df |>
    filter(!is.na(author_key), author_key != "") |>
    count(author_key, name = "n_comments", sort = TRUE)

  top_10_authors_share <- if (nrow(authors) > 0) {
    top_10_n <- max(1, ceiling(nrow(authors) * 0.10))
    100 * sum(head(authors$n_comments, top_10_n)) / sum(authors$n_comments)
  } else {
    NA_real_
  }

  valid_text <- df$text_original[
    !is.na(df$text_original) & nzchar(df$text_original)
  ]

  tibble(
    comments = n,
    reply_pct = round(mean(df$is_reply) * 100, 2),
    median_likes = median(likes),
    p90_likes = as.numeric(quantile(likes, 0.90)),
    p99_likes = as.numeric(quantile(likes, 0.99)),

    exact_duplicate_texts = sum(duplicated(valid_text)),
    duplicate_comment_ids = sum(duplicated(df$comment_id))
  )
}

Data Collection

In this assignment, we collected YouTube comments about the 2026 U.S. midterm elections using the YouTube Data API and the tuber package in R. We searched for “US midterm election” before (September 1–October 27) and after (November 4–collection date) Election Day. For each period, we searched by relevance and view count, collecting up to 100 videos per search.

The two search methods returned different results. Before the election, only 27 videos appeared in both lists, compared with 30 after the election. CNN appeared much more frequently in the view-count results, while the relevance results included a wider range of channels. Although YouTube reported about one million results, we could only retrieve a limited number of videos.

We selected three channels: CNN, NBC News, and FOX 9. We also checked their uploaded videos using the title filter midterm|election|senate|house race|ballot. However, the 500 most recent uploads did not reach September 1 for any of the channels. We then selected three videos per channel for each period, resulting in 18 videos. Comment collection was limited to 300 per video. We planned the API quota before collection and checked its usage afterward.

Comment Characteristics

The selected videos reported 61,583 comments, but our dataset contained 5,128 comments and replies: 2,428 before the election and 2,700 after. The number collected was much smaller than the reported total, partly because of the collection limit. Deleted or moderated comments may also explain some of the difference.We used parent comment IDs to recover missing video information for replies, allowing all 5,128 comments to be assigned to their corresponding videos.

About 31.6% of the comments before the election were replies, compared with 22.1% after the election. Most comments received very few likes, with a median of zero in both periods. However, likes were concentrated in a small number of comments. The top 1% received 42.1% of all likes before the election and 56.6% afterward.

We also examined how frequently users commented. The most active 10% of authors contributed around 26–27% of comments. There were 67 exact duplicate texts, although no duplicate comment IDs were found. These results suggest that some users and comments were more active or visible than others.

Text Analysis

We used individual comments as the unit of analysis. URLs and mentions were removed, and punctuation and symbols were handled during tokenization. Emojis were not separately analyzed.

Among the 100 examples, grievances were the most common category (45%), followed by election processes (30%), party references (17%), and candidate references (8%). We also examined phrases such as “early voting,” “election fraud,” “vote blue,” and “vote red.” The last two appeared 56 and 53 times, respectively.

Keyness analysis showed differences across channels. CNN comments included more references to Trump and international issues, while FOX 9 comments focused more on state-level politics. NBC News comments often mentioned Steve Kornacki and polling. We also found differences between the two election periods, although these may partly reflect the topics of the selected videos.

For sentiment analysis, we used the LSD2015 dictionary. On CNN, the share of negative comments increased from 35.7% before the election to 42.7% afterward. In contrast, positive comments increased on FOX 9 (23.1% to 36.8%) and NBC News (27.0% to 42.4%).

We compared the dictionary results with labels for 40 sampled comments. Agreement was only 47.5% (19 out of 40). Some negative political statements were classified as positive, while other comments were classified as neutral. This shows that dictionary-based sentiment analysis can miss negation, sarcasm, and political context.
