clean_cpd_body <- function(x) {

  if (is.na(x) || !nzchar(trimws(x))) {
    return(NA_character_)}

  lines <- strsplit(x, "\n", fixed = TRUE)[[1]]
  nonblank <- which(nzchar(trimws(lines)))

  if (length(nonblank) >= 4 &&
      grepl("^Administration of",  
            trimws(lines[nonblank[1]]))) 
    {lines <- lines[-seq_len(nonblank[3])]}

  x <- paste(lines, collapse = "\n")

  x <- sub("(?ims)^\\s*(?:N\\s*O\\s*T\\s*E|Categories|Names|Subjects|DCPD Number)\\s*:.*$", "", x, perl = TRUE)

  trimws(x)
}
