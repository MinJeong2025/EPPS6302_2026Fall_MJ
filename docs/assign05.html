3. Download Ten PDFs and Keep a Log

The GovInfo packages endpoint provides document metadata and links to PDF and other formats. Downloads are saved in a private folder, not the published website.

get_package_summary <- function(id) {
  request(paste0(
    "https://api.govinfo.gov/packages/",
    id, "/summary"
  )) |>
    req_url_query(api_key = key) |>
    req_perform() |>
    resp_body_json(simplifyVector = FALSE)
}

summaries <- purrr::map(
  selected$packageId,
  get_package_summary
)

get_pdf_url <- function(s, id) {

  api_url <- s$download$pdfLink

  if (!is.null(api_url) &&
      is.character(api_url) &&
      length(api_url) == 1 &&
      !is.na(api_url) &&
      grepl("^https://", api_url)) {
    return(api_url)
  }

  paste0("https://www.govinfo.gov/content/pkg/",id, "/pdf/", id, ".pdf")
}

pdf_urls <- purrr::map2_chr(summaries, selected$packageId, get_pdf_url)

safe_id <- gsub("[^A-Za-z0-9_-]",  "_",  selected$packageId)

log_tbl <- purrr::map_dfr(seq_len(nrow(selected)), function(i) {
  destination <- file.path(private_dir, paste0(safe_id[i], ".pdf"))

  status <- tryCatch({

    download.file(pdf_urls[i], destfile = destination, mode = "wb", quiet = TRUE)

    if (!file.exists(destination) ||
        file.size(destination) < 5) {
      stop("File missing or empty")
    }

    con <- file(destination, open = "rb")
    signature <- tryCatch(readBin(con, what = "raw", n = 5), finally = close(con))

    if (!identical(
      signature,
      charToRaw("%PDF-")
    )) {
      stop("Downloaded file is not a PDF")
    }
    "success"}, error = function(e) {
    if (file.exists(destination)) {unlink(destination)}
    paste("failed:", conditionMessage(e))})

  Sys.sleep(1)

  tibble(
    packageId = selected$packageId[i],
    title = selected$title[i],
    pdf_file = basename(destination),
    pdf_url = pdf_urls[i],
    status = status,
    downloaded_at = as.character(Sys.time())
  )
})

write_csv(log_tbl,"assign05_download_log.csv")
